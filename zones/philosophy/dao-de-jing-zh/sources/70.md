%[t2|甚]%[t4|易]%[t2|行]
shén yì háng

%[t1|天]%[t4|下]%[t4|莫]%[t2|能]%[t1|知]

%[t2|言]%[t3|有]%[t1|宗]

%[t4|事]%[t3|有]%[t1|君]

%[t1|夫]%[t2|唯]%[t2|無]%[t1|知]

%[t4|是]%[t3|以]%[t4|不]%[t3|我]%[t1|知]
